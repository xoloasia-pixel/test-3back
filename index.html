  <script>

    const CONFIG = {

      nBack: 3,

      practiceTrials: 10,

      practiceTargets: 3,

      mainTrials: 30,

      mainTargets: 10,

      stimDurationMs: 1500,

      isiDurationMs: 500,

      letters: ["B", "C", "D", "F", "G", "H", "J", "K", "L", "M", "N", "P", "R", "S", "T"],

      googleSheetsWebhookUrl: ""

    };


    let participantId = "anonim";

    let phase = "idle";

    let sequence = [];

    let currentIndex = 0;

    let stimStartTime = 0;

    let respondedThisTrial = false;

    let responseRtMs = null;

    let trialRecords = [];

    let stimTimer = null;

    let nextTrialTimer = null;


    const elIntro = document.getElementById("screen-intro");

    const elReady = document.getElementById("screen-ready");

    const elTrial = document.getElementById("screen-trial");

    const elResults = document.getElementById("screen-results");

    const elProgress = document.getElementById("progress-label");

    const elStimulus = document.getElementById("stimulus-area");

    const elFeedback = document.getElementById("feedback-area");

    const elSummary = document.getElementById("summary-table");

    const elSaveStatus = document.getElementById("save-status");

    const elIdInput = document.getElementById("participant-id");


    const urlParams = new URLSearchParams(window.location.search);

    if (urlParams.get("id")) {

      elIdInput.value = urlParams.get("id");

    }


    function showScreen(screenEl) {

      [elIntro, elReady, elTrial, elResults].forEach(s => s.classList.add("hidden"));

      screenEl.classList.remove("hidden");

    }


    function randomItem(arr, excludeArr = []) {

      const pool = arr.filter(x => !excludeArr.includes(x));

      return pool[Math.floor(Math.random() * pool.length)];

    }


    function shuffle(arr) {

      const copy = arr.slice();

      for (let i = copy.length - 1; i > 0; i--) {

        const j = Math.floor(Math.random() * (i + 1));

        [copy[i], copy[j]] = [copy[j], copy[i]];

      }

      return copy;

    }


    function generateSequence(totalTrials, targetCount, n) {

      const eligibleIndices = [];

      for (let i = n; i < totalTrials; i++) {

        eligibleIndices.push(i);

      }

      const chosenTargets = new Set(shuffle(eligibleIndices).slice(0, targetCount));


      const items = [];

      for (let i = 0; i < totalTrials; i++) {

        const isTarget = chosenTargets.has(i);

        let letter;

        if (isTarget) {

          letter = items[i - n].letter;

        } else {

          const forbidden = [];

          if (i >= n) forbidden.push(items[i - n].letter);

          letter = randomItem(CONFIG.letters, forbidden);

        }

        items.push({

          trialNumber: i + 1,

          letter,

          condition: isTarget ? "target" : "nontarget"

        });

      }

      return items;

    }


    function startBlock(selectedPhase) {

      clearTimeout(stimTimer);

      clearTimeout(nextTrialTimer);


      phase = selectedPhase;

      currentIndex = 0;

      trialRecords = [];


      const total = phase === "practice" ? CONFIG.practiceTrials : CONFIG.mainTrials;

      const targets = phase === "practice" ? CONFIG.practiceTargets : CONFIG.mainTargets;

      sequence = generateSequence(total, targets, CONFIG.nBack);


      showScreen(elTrial);

      runTrial();

    }


    function runTrial() {

      if (currentIndex >= sequence.length) {

        finishBlock();

        return;

      }


      const item = sequence[currentIndex];

      respondedThisTrial = false;

      responseRtMs = null;

      elFeedback.textContent = "";

      elFeedback.className = "feedback";

      elProgress.textContent =

        (phase === "practice" ? "[TRENING] " : "") +

        `Próba ${currentIndex + 1} / ${sequence.length}`;


      elStimulus.className = "stimulus";

      elStimulus.textContent = item.letter;

      stimStartTime = performance.now();


      stimTimer = setTimeout(() => {

        elStimulus.className = "fixation";

        elStimulus.textContent = "+";

      }, CONFIG.stimDurationMs);


      nextTrialTimer = setTimeout(() => {

        finalizeTrial(item);

      }, CONFIG.stimDurationMs + CONFIG.isiDurationMs);

    }


    function finalizeTrial(item) {

      const isTarget = item.condition === "target";

      const pressedSpace = respondedThisTrial;


      let outcome;

      let correct = 0;


      if (isTarget && pressedSpace) {

        outcome = "hit";

        correct = 1;

      } else if (isTarget && !pressedSpace) {

        outcome = "miss";

        correct = 0;

      } else if (!isTarget && pressedSpace) {

        outcome = "false_alarm";

        correct = 0;

      } else {

        outcome = "correct_rejection";

        correct = 1;

      }


      trialRecords.push({

        participantId,

        phase,

        trialNumber: item.trialNumber,

        letter: item.letter,

        condition: item.condition,

        responded: pressedSpace ? 1 : 0,

        rtMs: responseRtMs !== null ? Math.round(responseRtMs) : "",

        outcome,

        correct

      });


      if (phase === "practice") {

        if (outcome === "hit") {

          elFeedback.textContent = "Dobrze! To był cel (zgodny z literą sprzed 3 kroków).";

          elFeedback.classList.add("ok");

        } else if (outcome === "miss") {

          elFeedback.textContent = "Przegapiony cel — ta litera była 3 kroki wcześniej.";

          elFeedback.classList.add("err");

        } else if (outcome === "false_alarm") {

          elFeedback.textContent = "Fałszywy alarm — ta litera nie była 3 kroki wcześniej.";

          elFeedback.classList.add("err");

        } else {

          elFeedback.textContent = "Poprawny brak reakcji.";

          elFeedback.classList.add("ok");

        }

      }


      currentIndex += 1;

      runTrial();

    }


    function finishBlock() {

      const prevPhase = phase;

      phase = "idle";


      if (prevPhase === "practice") {

        showScreen(elReady);

        return;

      }


      const summary = computeSummary(trialRecords);

      renderSummary(summary);

      showScreen(elResults);

      sendToGoogleSheetsIfConfigured(summary, trialRecords);

    }


    function computeSummary(records) {

      const hits = records.filter(r => r.outcome === "hit").length;

      const misses = records.filter(r => r.outcome === "miss").length;

      const falseAlarms = records.filter(r => r.outcome === "false_alarm").length;

      const correctRejections = records.filter(r => r.outcome === "correct_rejection").length;

      const total = records.length || 1;

      const correctSum = records.reduce((acc, r) => acc + r.correct, 0);

      const accuracyPct = ((correctSum / total) * 100).toFixed(1);


      const rts = records

        .filter(r => r.responded === 1 && typeof r.rtMs === "number")

        .map(r => r.rtMs);

      const meanRtMs = rts.length

        ? Math.round(rts.reduce((a, b) => a + b, 0) / rts.length)

        : null;


      return {

        participantId,

        timestampIso: new Date().toISOString(),

        trialsTotal: records.length,

        hits,

        misses,

        falseAlarms,

        correctRejections,

        accuracyPct,

        meanRtMs: meanRtMs !== null ? meanRtMs : ""

      };

    }


    function renderSummary(s) {

      elSummary.innerHTML = `

        <tr><th>ID badanego</th><td>${escapeHtml(s.participantId)}</td></tr>

        <tr><th>Trafienia (Hits)</th><td>${s.hits}</td></tr>

        <tr><th>Pominięcia (Misses)</th><td>${s.misses}</td></tr>

        <tr><th>Fałszywe alarmy (False Alarms)</th><td>${s.falseAlarms}</td></tr>

        <tr><th>Poprawne odrzucenia (Correct Rejections)</th><td>${s.correctRejections}</td></tr>

        <tr><th>Dokładność (Accuracy)</th><td>${s.accuracyPct}%</td></tr>

        <tr><th>Średni czas reakcji (Mean RT)</th><td>${s.meanRtMs !== "" ? s.meanRtMs + " ms" : "brak reakcji"}</td></tr>

      `;

    }


    function escapeHtml(str) {

      return String(str).replace(/[&<>"']/g, m => ({

        "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"

      }[m]));

    }


    function buildCsv(records) {

      const headers = [

        "participant_id", "phase", "trial_number", "letter",

        "condition", "responded", "rt_ms", "outcome", "correct"

      ];

      const lines = [headers.join(",")];

      for (const r of records) {

        const row = [

          r.participantId, r.phase, r.trialNumber, r.letter,

          r.condition, r.responded, r.rtMs, r.outcome, r.correct

        ].map(val => `"${String(val).replace(/"/g, '""')}"`);

        lines.push(row.join(","));

      }

      return lines.join("\n");

    }


    function downloadCsv() {

      const csvContent = buildCsv(trialRecords);

      const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });

      const url = URL.createObjectURL(blob);

      const a = document.createElement("a");

      a.href = url;

      a.download = `3back_${participantId}_${Date.now()}.csv`;

      a.click();

      URL.revokeObjectURL(url);

    }


    async function sendToGoogleSheetsIfConfigured(summary, records) {

      if (!CONFIG.googleSheetsWebhookUrl) return;

      elSaveStatus.textContent = "Zapisywanie wyników do arkusza...";

      try {

        await fetch(CONFIG.googleSheetsWebhookUrl, {

          method: "POST",

          mode: "no-cors",

          headers: { "Content-Type": "application/json" },

          body: JSON.stringify({ summary, records })

        });

        elSaveStatus.textContent = "Wyniki zostały zapisane.";

      } catch (err) {

        elSaveStatus.textContent = "Nie udało się wysłać automatycznie — pobierz plik CSV przyciskiem poniżej.";

      }

    }


    document.addEventListener("keydown", (e) => {

      if (phase !== "practice" && phase !== "main") return;

      if (e.code === "Space" || e.key === " ") {

        e.preventDefault();

        if (!respondedThisTrial) {

          respondedThisTrial = true;

          responseRtMs = performance.now() - stimStartTime;

        }

      }

    });


    document.getElementById("btn-start-practice").addEventListener("click", () => {

      const val = elIdInput.value.trim();

      participantId = val || "anonim";

      startBlock("practice");

    });


    document.getElementById("btn-start-main").addEventListener("click", () => {

      startBlock("main");

    });


    document.getElementById("btn-download-csv").addEventListener("click", downloadCsv);


    document.getElementById("btn-restart").addEventListener("click", () => {

      showScreen(elIntro);

    });
